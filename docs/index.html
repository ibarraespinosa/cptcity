NOAA — 14 ramps for Earth science

Deep navy to sky blue to white — plus storm, radar, satellite, and climate ramps for NWS forecasting, hurricanes, NEXRAD, GOES, buoys, and fisheries.

library(cptcity)
find_cpt("noaa")
# 14 ramps: noaa, noaa_storm, noaa_nws, noaa_nhc,
# noaa_nexrad, noaa_goes, noaa_buoy, noaa_ncei,
# noaa_nmfs, noaa_jetstream, noaa_tornado,
# noaa_wind_chill, noaa_heat_index, noaa_coastal
cols <- cpt("noaa_nhc", n = 256)

Installation

Base R + grDevices only. ● tested with testthat

# install
install.packages("remotes")
remotes::install_github("ibarraespinosa/cptcity")

# update
remotes::install_github("ibarraespinosa/cptcity", upgrade = "always")

library(cptcity)
cpt("noaa", n = 256)
# Python sibling package (pycptcity)
# pip install git+https://github.com/ibarraespinosa/pycptcity
from pycptcity import cpt
cpt("noaa", n = 256)
